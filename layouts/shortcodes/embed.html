{{/* Embed a standalone HTML page (e.g. a Claude artifact) that sits in this page's bundle. Shown as a preview; "Expand" fills the window, and closing (✕, Esc, or the browser's back button) returns you to where you were. {{< embed src="acts-in-motion.embed.html" title="Acts in Motion" preview="contention.png" >}} preview a screenshot in the bundle, shown instead of the live page. The page itself is then only loaded on the first expand. height without a preview, the live page is shown in a box this tall (420px) Files must end in `.embed.html` -- config.toml ignores those as content, and this shortcode publishes them alongside the page as `.html`. (claude.ai artifact URLs can't be iframed: they send frame-ancestors 'self'.) */}} {{- $src := .Get "src" -}} {{- $title := .Get "title" | default "Embedded page" -}} {{- $height := .Get "height" | default "420px" -}} {{- $file := path.Join "content" .Page.File.Dir $src -}} {{- if not (fileExists $file) -}} {{- errorf "embed: %q not found (in %s)" $src .Position -}} {{- end -}} {{- $target := path.Join (strings.TrimPrefix "/" .Page.RelPermalink) (replace $src ".embed.html" ".html") -}} {{- $page := resources.FromString $target (readFile $file) -}} {{- $preview := false -}} {{- with .Get "preview" -}} {{- $preview = $.Page.Resources.Get . -}} {{- if not $preview -}} {{- errorf "embed: preview %q not found (in %s)" . $.Position -}} {{- end -}} {{- end -}} {{- if not (.Page.Store.Get "embedScript") -}} {{- .Page.Store.Set "embedScript" true -}} {{- end -}}
